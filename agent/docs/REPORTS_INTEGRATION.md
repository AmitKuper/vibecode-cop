# Integrating Report Generation into Game Orchestrator

## Where to Call Reports

After a game ends and game state is finalized, call the report system:

```python
# agent/orchestrator.py
# In the method that handles game completion

async def finalize_game(self, game_id: str, game_state: dict):
    """Finalize game and generate reports."""
    
    # ... existing game-end logic ...
    
    # NOW: Generate reports
    await self.generate_reports(game_id, game_state)


async def generate_reports(self, game_id: str, game_state: dict):
    """Generate all reports (file + Gmail)."""
    from agent.reports.bundle import ReportBundleBuilder
    from agent.reports.plugin_factory import ReportPluginFactory
    from agent.reports.manager import ReportManager
    
    try:
        logger.info(f"Generating reports for {game_id}...")
        
        # Build report context
        context = await ReportBundleBuilder(
            self.games_dir / game_id  # Path to game's memory directory
        ).build(
            game_id=game_id,
            role=self.role,  # "cop" or "thief"
            game_state=game_state,
            result={
                "winner": game_state.get("winner"),
                "step": game_state.get("step", 0),
            },
            config_hash=self.config_sha256,  # SHA-256 of game_config.toml
            log_hash=None,  # Optional: SHA-256 of game moves log
            metadata={
                "group_id": getattr(self, "group_id", "unknown"),
                "opponent_group_id": getattr(self, "opponent_group_id", None),
            },
        )
        
        # Load plugins from config
        reports_config = self.config.get("reports", {})
        plugins = await ReportPluginFactory.from_config(reports_config)
        
        if not plugins:
            logger.warning(f"No report plugins configured")
            return
        
        # Execute all plugins (isolated failures)
        results = await ReportManager(plugins).generate_all(context)
        
        # Log results
        for plugin_name, result in results.items():
            if result.ok:
                logger.info(
                    f"Report [{plugin_name}] {result.status}: {result.destination}"
                )
            else:
                logger.error(
                    f"Report [{plugin_name}] FAILED: {result.error_code} — {result.error}"
                )
        
        logger.info(f"Report generation complete for {game_id}")
        
    except Exception as e:
        logger.error(f"Report generation failed: {e}", exc_info=True)
        # Don't crash — game state is already saved
```

## Config Loading

Make sure the orchestrator loads the reports config:

```python
# In __init__ or setup method

import tomllib
from pathlib import Path

def load_config(config_path: str | Path):
    """Load agent config."""
    with open(config_path, 'rb') as f:
        config = tomllib.load(f)
    
    return config

# In __init__:
self.config = load_config("cop/config.toml")  # or "thief/config.toml"

# Later, when reporting:
reports_config = self.config.get("reports", {})
```

## Directory Structure Expected

The orchestrator should have access to:

```
agent/memory/
└── {game_id}/
    ├── game_state.json           ← Final game state
    ├── moves.jsonl               ← Move history (optional)
    ├── declaration_{game_id}.json ← Created by bundle builder if missing
    ├── config_{game_id}_g00.json  ← Created by bundle builder if missing
    ├── log_{game_id}_g00.json     ← Created by bundle builder if missing
    ├── result_{game_id}.json      ← Created by bundle builder if missing
    ├── report.json               ← Generated by FileReportPlugin
    ├── report.md                 ← Generated by FileReportPlugin
    └── report_delivery.json      ← Generated by GmailReportPlugin (idempotency)
```

The `ReportBundleBuilder` will create any missing required files as stubs.

## Minimal Example

```python
# Minimal code to add to orchestrator

async def handle_game_end(self):
    """Called when game loop exits."""
    
    # Save game state (already done elsewhere)
    # ...
    
    # Generate reports
    from agent.reports.bundle import ReportBundleBuilder
    from agent.reports.plugin_factory import ReportPluginFactory
    from agent.reports.manager import ReportManager
    
    context = await ReportBundleBuilder(
        Path("agent/memory") / self.game_id
    ).build(
        game_id=self.game_id,
        role=self.role,
        game_state=self.game_state,
        result={"winner": self.game_state.get("winner")},
        config_hash=self.config_sha256,
        metadata={"group_id": self.group_id},
    )
    
    plugins = await ReportPluginFactory.from_config(
        self.config.get("reports", {})
    )
    
    results = await ReportManager(plugins).generate_all(context)
    logger.info(f"Reports: {results}")
```

## Testing Report Generation

To test without running full agents:

```python
# test_reports_integration.py
import asyncio
from pathlib import Path
from agent.reports.bundle import ReportBundleBuilder
from agent.reports.plugin_factory import ReportPluginFactory
from agent.reports.manager import ReportManager

async def test_report_generation():
    """Test report generation in isolation."""
    
    # Create test game directory
    game_dir = Path("agent/memory/test_game_001")
    game_dir.mkdir(parents=True, exist_ok=True)
    
    # Fake game state
    game_state = {
        "game_id": "test_game_001",
        "winner": "cop",
        "step": 10,
        "cop_position": [3, 3],
        "thief_position": [3, 3],
        "created_at": "2026-07-20T00:00:00",
        "ended_at": "2026-07-20T00:05:00",
        "move_history": [],
    }
    
    # Build context
    context = await ReportBundleBuilder(game_dir).build(
        game_id="test_game_001",
        role="cop",
        game_state=game_state,
        result={"winner": "cop"},
        config_hash="test_hash",
        metadata={"group_id": "test_group"},
    )
    
    # Load plugins (with default test config)
    plugins = await ReportPluginFactory.from_config({
        "enabled": True,
        "plugins": ["file_json", "file_markdown"],
        "file_json": {"enabled": True},
        "file_markdown": {"enabled": True},
        "gmail": {"enabled": False},  # Skip Gmail for testing
    })
    
    # Generate reports
    results = await ReportManager(plugins).generate_all(context)
    
    # Check results
    print(f"✓ Generated {len(results)} reports:")
    for name, result in results.items():
        print(f"  - {name}: {result.status} → {result.destination}")
    
    # Verify files exist
    assert (game_dir / "report.json").exists()
    assert (game_dir / "report.md").exists()
    print("\n✓ All report files created successfully")

if __name__ == "__main__":
    asyncio.run(test_report_generation())
```

Run with:
```bash
python agent/docs/test_reports_integration.py
```

## Enabling Gmail in Stages

### Stage 1: Dry-run (Safe, No Credentials Needed)

```toml
[reports.gmail]
mode = "dry_run"  # Saves local .txt preview
```

### Stage 2: Draft (Creates .eml files)

```toml
[reports.gmail]
mode = "draft"  # Creates .eml files you can inspect
```

### Stage 3: Send (Real Gmail)

```toml
[reports.gmail]
mode = "send"  # Requires OAuth token.json
```

## Error Handling

Reports should **never crash the game**:

```python
try:
    await self.generate_reports(game_id, game_state)
except Exception as e:
    logger.error(f"Report generation failed (game saved): {e}")
    # Game continues — state is already persisted
```

Plugin failures are isolated by `ReportManager`, so one plugin crash doesn't affect others.
